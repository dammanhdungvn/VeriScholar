% 7. Phân cấp đề mục và độ sâu mục lục (Tối đa 4 cấp)

\setcounter{secnumdepth}{4}
\setcounter{tocdepth}{3}

% Định dạng tiêu đề chương chuẩn 16pt in hoa đậm căn giữa
% Dạng block: CHƯƠNG X: TÊN CHƯƠNG (chương có số) hoặc TÊN CHƯƠNG (chương không số)
\titleformat{\chapter}[block]
  {\fontsize{16}{20}\selectfont\bfseries\centering}
  {\MakeUppercase{\chaptertitlename}\ \thechapter:\ }
  {0pt}
  {\MakeUppercase}
\titlespacing*{\chapter}{0pt}{-10pt}{18pt}

% Mục cấp 1 (1.1, 2.1...): 14pt Đậm, căn trái
\titleformat{\section}
  {\fontsize{14}{18}\selectfont\bfseries}
  {\thesection}{0.75em}{}
\titlespacing*{\section}{0pt}{12pt}{6pt}

% Tiểu mục cấp 2 (1.1.1, 2.1.1...): 13pt Đậm, căn trái

  {\fontsize{13}{17}\selectfont\bfseries}
  {\thesubsection}{0.75em}{}
\titlespacing*{\subsection}{0pt}{9pt}{4pt}

% Tiểu mục cấp 3 (1.1.1.1, 2.1.1.1...): 13pt Đậm nghiêng, căn trái

  {\fontsize{13}{17}\selectfont\bfseries\itshape}
