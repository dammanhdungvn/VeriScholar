% Tên viết tắt chuẩn hóa của dự án
\newcommand{\VS}{\textbf{VeriScholar}}

% Ký hiệu toán học công thức RRF
\newcommand{\RRF}{\text{RRF}}

% Macro định dạng đoạn trích dẫn quan trọng
\newcommand{\academicquote}[1]{%
    \begin{quote}
        \itshape #1
    \end{quote}
}

% Macro ghi chú thuật ngữ viết tắt lần đầu
\newcommand{\firstacronym}[2]{\textit{#1} (#2)}

% Chuẩn hóa định dạng chữ LaTeX & TeX thành văn bản thông thường (Times New Roman),
% loại bỏ triệt để hiệu ứng nâng hạ ký tự (thò thụt) và phông chữ Computer Modern mặc định của logo LaTeX
\usepackage{xspace}
\renewcommand{\LaTeX}{LaTeX\xspace}
\renewcommand{\TeX}{TeX\xspace}
