\section{Thiết kế cơ sở dữ liệu vật lý}
\label{sec:thiet_ke_db}

\subsection{Lược đồ quan hệ thực thể (ERD)}
\label{subsec:erd}
Hệ thống \VS\ sử dụng cơ sở dữ liệu quan hệ PostgreSQL 16 kết hợp tiện ích mở rộng \texttt{pgvector}. Cấu trúc dữ liệu bao gồm 16 bảng vật lý chuẩn hóa, được phân nhóm khoa học thành 4 cụm tri thức học thuật logic, bảo đảm tính toàn vẹn giao dịch ACID tuyệt đối và phân hoạch an toàn theo định danh người dùng:

\begin{enumerate}[label=\textbf{(\arabic*)}]
    \item \textbf{Cụm Định danh học giả và Két sắt cá nhân (3 bảng):} \texttt{users} (thông tin tài khoản và mã định danh quốc tế ORCID iD), \texttt{email\_otps} (quản lý phiên đăng nhập không mật khẩu) và \texttt{user\_secrets} (két sắt lưu trữ khóa API cá nhân BYOK mã hóa AES-256-GCM AEAD).
    \item \textbf{Cụm Ấn phẩm và Đoạn trích (4 bảng):} Gồm \texttt{documents}, \texttt{storage\_blobs}, \texttt{document\_chunks} và \texttt{discovery\_cache} lưu trữ hồ sơ, kho tệp CAS và đoạn trích tọa độ kèm vector 1536 chiều.
    \item \textbf{Cụm Đồ thị tri thức học thuật 4 thực thể (4 bảng):} \texttt{paper\_nodes} (thực thể bài báo), \texttt{method\_nodes} (thực thể phương pháp và kỹ thuật đề xuất), \texttt{dataset\_nodes} (thực thể bộ dữ liệu chuẩn benchmark) và \texttt{metric\_results} (thực thể kết quả đo lường định lượng kèm đơn vị).
    \item \textbf{Cụm Phiên đọc sâu, Ma trận và Bản thảo (5 bảng):} \texttt{reading\_sessions} (phiên nghiên cứu chuyên sâu của học giả), \texttt{chat\_messages} (lịch sử hỏi đáp kèm huy hiệu trích dẫn tọa độ), \texttt{comparison\_matrices} (bảng đối chiếu so sánh đa phương pháp), \texttt{draft\_projects} (dự án soạn thảo bài báo) và \texttt{draft\_files} (cây thư mục tệp mã nguồn \LaTeX).
\end{enumerate}
